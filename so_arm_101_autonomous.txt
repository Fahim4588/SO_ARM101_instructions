import cv2
import rerun as rr

# Initialize Rerun
rr.init("camera_feedback", spawn=True)

# Open camera
cap = cv2.VideoCapture(0)

while True:
    ret, frame = cap.read()

    if not ret:
        break

    # OpenCV uses BGR; Rerun expects RGB
    frame_rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)

    # Log the camera frame to Rerun
    rr.log("camera", rr.Image(frame_rgb))

    # Optional: also show using OpenCV
    cv2.imshow("Camera", frame)

    if cv2.waitKey(1) & 0xFF == ord("q"):
        break

cap.release()
cv2.destroyAllWindows()
